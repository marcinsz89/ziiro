cmake_minimum_required(VERSION 3.20)
project(ziiro VERSION 1.1.0 LANGUAGES C CXX)

option(ZIIRO_DEVICE "Build the ARM64 Linux Cardputer Zero app" OFF)
if(ZIIRO_DEVICE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "The device app requires an ARM64 Linux toolchain.")
    endif()
    add_compile_definitions(ZIIRO_DEVICE)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
endif()

set(LVGL_SOURCE_DIR "" CACHE PATH "Use an existing LVGL 9.5.0 checkout")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE FILEPATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
include(FetchContent)

if(LVGL_SOURCE_DIR)
    add_subdirectory("${LVGL_SOURCE_DIR}" lvgl EXCLUDE_FROM_ALL)
    set(LVGL_LICENSE "${LVGL_SOURCE_DIR}/LICENCE.txt")
else()
    FetchContent_Declare(lvgl
        GIT_REPOSITORY https://github.com/lvgl/lvgl.git
        GIT_TAG 85aa60d18b3d5e5588d7b247abf90198f07c8a63
    )
    FetchContent_MakeAvailable(lvgl)
    set(LVGL_LICENSE "${lvgl_SOURCE_DIR}/LICENCE.txt")
endif()

if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
# Fetch sources without adding the upstream CMake project until its tools are disabled.
FetchContent_Declare(webp
    URL https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz
    URL_HASH SHA256=e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564
    SOURCE_SUBDIR src
)
FetchContent_MakeAvailable(webp)
# Build the decoder as an app dependency without installing libwebp's tools or libraries.
foreach(tool ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${tool} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${webp_SOURCE_DIR}" "${webp_BINARY_DIR}" EXCLUDE_FROM_ALL)

set(ZIIRO_SOURCES studio.cpp catalog.cpp archive_artwork.cpp artwork_convert.c nfc.c)
set(ZIIRO_ICON packaging/ziiro-icon-100.png)
set(ZIIRO_PLATFORM_NOTICE assets/Platform-logos.txt)
if(ZIIRO_DEVICE)
    set(ZIIRO_TARGET ziiro)
    add_executable(ziiro device.c ${ZIIRO_SOURCES})
    target_sources(ziiro PRIVATE third_party/amiibo-zero/ntag215.c third_party/amiibo-zero/st25r3916.c)
    target_include_directories(ziiro PRIVATE third_party/amiibo-zero)
    install(TARGETS ziiro RUNTIME DESTINATION bin)
    install(FILES tools/sync_catalog.py DESTINATION share/ziiro)
    install(FILES assets/game-logos.json DESTINATION share/ziiro)
    install(FILES assets/nfc-presenting.wav DESTINATION share/ziiro)
    install(FILES "${ZIIRO_ICON}" DESTINATION share/ziiro RENAME ziiro-icon.png)
    install(FILES "${ZIIRO_PLATFORM_NOTICE}" THIRD_PARTY_NOTICES.md DESTINATION share/doc/ziiro)
    install(FILES packaging/ziiro.desktop DESTINATION share/APPLaunch/applications)
    install(FILES assets/AmiiboAPI-LICENSE.txt DESTINATION share/doc/ziiro)
    install(FILES "${LVGL_LICENSE}" DESTINATION share/doc/ziiro RENAME LVGL-LICENSE.txt)
    install(FILES "${webp_SOURCE_DIR}/COPYING" DESTINATION share/doc/ziiro RENAME libwebp-LICENSE.txt)
    install(FILES "${webp_SOURCE_DIR}/PATENTS" DESTINATION share/doc/ziiro RENAME libwebp-PATENTS.txt)
    install(FILES third_party/amiibo-zero/LICENSE DESTINATION share/doc/ziiro RENAME amiibo-zero-LICENSE.txt)
    if(ZIIRO_ZIG)
        get_filename_component(ZIIRO_ZIG_REAL "${ZIIRO_ZIG}" REALPATH)
        get_filename_component(ZIIRO_ZIG_ROOT "${ZIIRO_ZIG_REAL}" DIRECTORY)
        foreach(runtime libcxx libcxxabi libunwind)
            install(FILES "${ZIIRO_ZIG_ROOT}/lib/${runtime}/LICENSE.TXT"
                    DESTINATION share/doc/ziiro RENAME "${runtime}-LICENSE.txt" OPTIONAL)
        endforeach()
        install(FILES "${ZIIRO_ZIG_ROOT}/LICENSE"
                DESTINATION share/doc/ziiro RENAME Zig-LICENSE.txt OPTIONAL)
    endif()

    configure_file(packaging/control.in debian-control @ONLY)
else()
    set(ZIIRO_TARGET ziiro-preview)
    add_executable(ziiro-preview preview.c ${ZIIRO_SOURCES})
    target_link_libraries(ziiro-preview PRIVATE PkgConfig::SDL2)
    add_executable(ziiro-export-artwork tools/export_artwork.cpp catalog.cpp archive_artwork.cpp)
    target_include_directories(ziiro-export-artwork PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(ziiro-export-artwork PRIVATE cxx_std_17)
    target_compile_options(ziiro-export-artwork PRIVATE -Wall -Wextra -Wpedantic)
    configure_file(tools/sync_catalog.py sync_catalog.py COPYONLY)
    configure_file(assets/game-logos.json game-logos.json COPYONLY)
    configure_file(assets/AmiiboAPI-LICENSE.txt AmiiboAPI-LICENSE.txt COPYONLY)
    configure_file("${ZIIRO_ICON}" ziiro-icon.png COPYONLY)

    install(TARGETS ziiro-preview RUNTIME DESTINATION ziiro)
    install(FILES tools/sync_catalog.py assets/game-logos.json assets/AmiiboAPI-LICENSE.txt DESTINATION ziiro)
    install(FILES "${ZIIRO_ICON}" DESTINATION ziiro RENAME ziiro-icon.png)
    install(FILES "${ZIIRO_PLATFORM_NOTICE}" THIRD_PARTY_NOTICES.md DESTINATION ziiro)
    install(FILES "${LVGL_LICENSE}" DESTINATION ziiro RENAME LVGL-LICENSE.txt)
    install(FILES "${webp_SOURCE_DIR}/COPYING" DESTINATION ziiro RENAME libwebp-LICENSE.txt)
    install(FILES "${webp_SOURCE_DIR}/PATENTS" DESTINATION ziiro RENAME libwebp-PATENTS.txt)
endif()

target_compile_features(${ZIIRO_TARGET} PRIVATE c_std_11 cxx_std_17)
target_compile_options(${ZIIRO_TARGET} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${ZIIRO_TARGET} SYSTEM PRIVATE "${webp_SOURCE_DIR}/src")
target_link_libraries(${ZIIRO_TARGET} PRIVATE lvgl::lvgl webpdecoder m)
